    97→// ---------- minimal but valid PDF generator ----------
    98→function escapePdfText(s) {
    99→  return s.replace(/\\/g, '\\\\').replace(/\(/g, '\\(').replace(/\)/g, '\\)').replace(/[^\x20-\x7E]/g, ' ');
   100→}
   101→
   102→function makeSeedPdf(title, lines) {
   103→  const contentParts = ['BT /F1 16 Tf 50 800 Td (' + escapePdfText(title) + ') Tj ET'];
   104→  let y = 770;
   105→  for (const line of lines) {
   106→    contentParts.push(`BT /F1 10 Tf 50 ${y} Td (${escapePdfText(line)}) Tj ET`);
   107→    y -= 16;
   108→  }
   109→  const content = contentParts.join('\n');
   110→  const objects = [];
   111→  objects.push('<< /Type /Catalog /Pages 2 0 R >>');
   112→  objects.push('<< /Type /Pages /Kids [3 0 R] /Count 1 >>');
   113→  objects.push('<< /Type /Page /Parent 2 0 R /MediaBox [0 0 595 842] /Contents 4 0 R /Resources << /Font << /F1 5 0 R >> >> >>');
   114→  objects.push(`<< /Length ${content.length} >>\nstream\n${content}\nendstream`);
   115→  objects.push('<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica >>');
   116→
   117→  let pdf = '%PDF-1.4\n';
   118→  const offsets = [];
   119→  objects.forEach((body, i) => {
   120→    offsets.push(pdf.length);
   121→    pdf += `${i + 1} 0 obj\n${body}\nendobj\n`;
   122→  });
   123→  const xrefStart = pdf.length;
   124→  pdf += `xref\n0 ${objects.length + 1}\n0000000000 65535 f \n`;
   125→  for (const off of offsets) {
   126→    pdf += `${String(off).padStart(10, '0')} 00000 n \n`;
   127→  }
   128→  pdf += `trailer\n<< /Size ${objects.length + 1} /Root 1 0 R >>\nstartxref\n${xrefStart}\n%%EOF`;
   129→  return Buffer.from(pdf, 'latin1');
   130→}

   135→function writeSeedFile(propertyDir, fileName, buffer) {
   136→  const dir = path.join(uploadRoot, 'seed', propertyDir);
   137→  fs.mkdirSync(dir, { recursive: true });
   138→  const absolute = path.join(dir, fileName);
   139→  fs.writeFileSync(absolute, buffer);
   140→  return path.relative(uploadRoot, absolute);
   141→}

   143→async function upsertUser({ email, password }) {
   144→  const normalized = String(email).trim().toLowerCase();
   145→  const name = NAME_BY_EMAIL[normalized];
   146→  const role = ROLE_BY_EMAIL[normalized];
   147→  if (!name || !role) throw new Error(`seed: email ${normalized} is not on the demo allowlist`);
   148→
   149→  const existing = await User.findOne({ email: normalized }).lean();
   150→  if (existing) {
   151→    // Deterministic alignment: identity + role follow the allowlist; the
   152→    // password hash is preserved so admin/user password changes survive.
   153→    if (existing.full_name !== name || existing.role !== role) {
   154→      await User.updateOne({ _id: existing._id }, { $set: { full_name: name, role } });
   155→    }
   156→    return User.findOne({ email: normalized }).lean();
   157→  }
   158→  const hash = await bcrypt.hash(password, 10);
   159→  await User.create({
   160→    full_name: name,
   161→    email: normalized,
   162→    password_hash: hash,
   163→    phone: PHONE_BY_EMAIL[normalized] || null,
   164→    district: PLACE_BY_EMAIL[normalized].district,
   165→    state: PLACE_BY_EMAIL[normalized].state,
   166→    wallet_address: WALLET_BY_EMAIL[normalized] || null,
   167→    role,
   168→    last_activity_at: new Date(Date.now() - 3 * 3600000),
   169→  });
   170→  return User.findOne({ email: normalized }).lean();
   171→}

   217→async function insertDocument({ property, uploadedBy, documentType, fileName, title, lines, verifiedBy, status, daysAgo }) {
   218→  const buffer = makeSeedPdf(title, lines);
   219→  const sha = sha256Buffer(buffer);
   220→  const storagePath = writeSeedFile(property.property_id, fileName, buffer);
   221→  const createdAt = new Date(Date.now() - daysAgo * 86400000);
   222→  const verifiedAt = verifiedBy ? new Date(Date.now() - (daysAgo - 1) * 86400000) : null;
   223→  await Document.create({
   224→    property_id: property._id,
   225→    uploaded_by: uploadedBy._id,
   226→    document_type: documentType,
   227→    file_name: fileName,
   228→    file_size: buffer.length,
   229→    mime_type: 'application/pdf',
   230→    storage_path: storagePath,
   231→    sha256_hash: sha,
   232→    verification_status: status,
   233→    verified_by: verifiedBy ? verifiedBy._id : null,
   234→    verified_at: verifiedAt,
   235→    created_at: createdAt,
   236→    updated_at: createdAt,
   237→  });
   238→  return Document.findOne({ property_id: property._id, file_name: fileName }).lean();
   239→}
